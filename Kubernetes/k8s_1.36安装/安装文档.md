### k8s 1.36安装手册

##### 主机名设置
```shell
hostnamectl set-hostname k8s-master01
```
##### ip地址设置
```shell
# 新建netplan的yaml文件
vi /etc/netplan/ens33-installer-config.yaml
network:
  version: 2
  ethernets:
    ens33:
      dhcp4: false
      dhcp6: true
      addresses:
        - 192.168.88.139/24
      routes:
        - to: default
        via: 192.168.88.2
      nameservers:
        addresses: [119.29.29.29, 8.8.8.8, 8.8.4.4]
      match:
        macaddress: 00:0c:29:2c:80:3a
      set-name: ens33
sudo chown root:root /etc/netplan/ens33-installer-config.yaml
sudo chmod 600 /etc/netplan/ens33-installer-config.yaml
sudo netplan apply
```

###### 配置内核转发及网桥过滤
```shell
# 创建加载内核模块文件
cat << EOF | tee /etc/modules-load.d/k8s.conf
overlay
br_netfilter
EOF

# 本次执行，手动加载此模块
modprobe overlay
modprobe br_netfilter

# 查看已加载的模块
lsmod | egrep "overlay"
lsmod | egrep "br_netfilter"

# 添加网桥过滤及内核转发配置文件
cat << EOF | tee /etc/sysctl.d/k8s.conf
net.bridge.bridge-nf-call-ip6tables = 1
net.bridge.bridge-nf-call-iptables = 1
net.ipv4.ip_forward = 1
EOF

#加载内核参数
sysctl --system
```

###### 关闭swap分区
```shell
swapoff -a && sed -i '/swap/s/^/#/' /etc/fstab
# 查看swap分区
free -mh
# 如果有swap分区则在/etc/fstab里将swap分区注释掉
```

###### 生成containerd配置文件
```shell
mkdir -p /etc/containerd
containerd config default > /etc/containerd/config.toml

# config.toml默认配置文件创建成功后，修改沙箱镜像
sed -i "s#registry.k8s.io/pause:3.8#registry.k8s.io/pause:3.10.2#g" /etc/containerd/config.toml
# 还需要修改驱动
sed -i "SystemdCgroup = false#SystemdCgroup = true#g" /etc/containerd/config.toml
```

###### k8s集群软件apt源准备

###### 下载用于Kubernetes软件包仓库的公共签名密钥
```shell
curl -fsSL https://pkgs.k8s.io/core:/stable:/v1.36/deb/Release.key | sudo gpg --dearmor -o /etc/apt/keyrings/kubernetes-apt-keyring.gpg
# 验证gpg key是否存在
ls /etc/apt/keyrings/
```

###### 添加Kubernetes仓库
```shell
echo 'deb [signed-by=/etc/apt/keyrings/kubernetes-apt-keyring.gpg] https://pkgs.k8s.io/core:/stable:/v1.36/deb/ /' | sudo tee /etc/apt/sources.list.d/kubernetes.list
# 验证gpg key是否存在
ls /etc/apt/sources.list.d/

# 更新Kubernetes仓库
apt-get update

# 验证Kubernetes仓库
apt-cache policy kubeadm kubelet kubectl cri-tools kubernetes-cni
```

###### 在master01上初始化kubernetes集群
```shell
# 修改kubernetes API域名地址映射
sed -i "192.168.88.100 www.k8s.org#192.168.88.134 www.k8s.org#g" /etc/hosts 

kubeadm init --control-plane-endpoint="www.k8s.org" \
             --kubernetes-version=v1.36.6 \
             --pod-network-cidr=10.244.0.0/16 \
             --service-cidr=10.96.0.0/12 \
             --token-ttl=0 \
             --upload-certs \
             --image-repository=registry.k8s.io
```

###### 在master01上生成kube-vip static pod
```shell
# 设置环境变量
export VIP=192.168.88.100
export INTERFACE=ens33
export KVVERSION=v1.3.1

alias kube-vip="ctr image pull ghcr.io/kube-vip/kube-vip:$KVVERSION; ctr run --rm --net-host ghcr.io/kube-vip/kube-vip:$KVVERSION vip /kube-vip"

kube-vip manifest pod \
    --interface $INTERFACE \
    --address $VIP \
    --controlplane \
    --services \
    --arp \
    --leaderElection | tee /etc/kubernetes/manifests/kube-vip.yaml

#将kube-vip.yaml复制到k8s-master02\k8s-master03上
scp /etc/kubernetes/manifests/kube-vip.yaml k8s-master02:/etc/kubernetes/manifests/kube-vip.yaml
scp /etc/kubernetes/manifests/kube-vip.yaml k8s-master03:/etc/kubernetes/manifests/kube-vip.yaml

```
